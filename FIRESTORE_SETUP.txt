#!/bin/bash
# Firestore Setup Instructions for Windows PowerShell

# Step 1: Get Firebase Service Account Key
# Go to: https://console.firebase.google.com/
# Project: datasell-7b993 (or your project)
# Go to: Project Settings (gear icon) > Service Accounts
# Click "Generate New Private Key"
# Save the JSON file

# Step 2: Create .env.local entry
# Add to .env.local:
# GOOGLE_APPLICATION_CREDENTIALS="path/to/your/service-account-key.json"

# Step 3: Enable Firestore in Firebase Console
# Go to: https://console.firebase.google.com/
# Project: datasell-7b993
# Firestore Database > Create Database
# Start in Production Mode (restrict read/write with security rules later)
# Select region: Any region (or closest to you)

# Step 4: Initialize Collections and Documents
# Navigate to Firestore and create these collections manually:
# - agents
# - packages
# - orders  
# - transactions

# Step 5: Run seeding script (after implementing)
# npx ts-node prisma/firestore-seed.ts

echo "Firestore setup guide created. Follow instructions above."
